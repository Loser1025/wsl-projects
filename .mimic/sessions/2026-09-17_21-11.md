- **実装内容**: 新たに `parsePoketeruCompletedCsv_()` 関数を追加しました。
  - 「ぽけてる予約一覧」CSVを読み込み、ヘッダー行から `ステータス` と `顧客名` の列を自動特定します。
  - **ステータスが「完了」**である行のみを抽出し、顧客名を `normalizeName_()` で正規化した上でインデックス（マップ）として保持します。

```javascript
/** ぽけてる予約一覧CSVを読み込み、ステータスが「完了」の案件の顧客名をマップ化する */
function parsePoketeruCompletedCsv_(csvText) {
  const rows = Utilities.parseCsv(csvText);
  if (rows.length < 2) return {};

  const header = rows[0];
  const statusIdx = header.indexOf('ステータス');
  const nameIdx = header.indexOf('顧客名');
  if (statusIdx === -1 || nameIdx === -1) {
    throw new Error('ぽけてるCSVに「ステータス」または「顧客名」列が見つかりません');
  }

  const completedMap = {};
  for (let i = 1; i < rows.length; i++) {
    const row = rows[i];
    const status = String(row[statusIdx] || '').trim();
    if (status === '完了') {
      const name = normalizeName_(row[nameIdx]);
      if (name) {
        completedMap[name] = row;
      }
    }
  }
  return completedMap;
}
```
